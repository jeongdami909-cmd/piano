gameArea.addEventListener("pointerdown", (e) => { // 1. 모바일 터치 간섭 및 스크롤 방지 e.preventDefault(); if (!isPlaying || activeRows.length === 0) return; // 🔥 V22 엔진: 배열 순서가 아니라 실제 화면의 'bottom' 위치로 맨 아래 줄을 찾음 const bottomRow = activeRows.reduce((lowest, r) => { const currentBottom = parseFloat(r.rowEl.style.bottom); const lowestBottom = parseFloat(lowest.rowEl.style.bottom); return currentBottom < lowestBottom ? r : lowest; }); // 중복 클릭 방지 if (bottomRow.locked) return; const rect = gameArea.getBoundingClientRect(); const touchX = e.clientX - rect.left; // 레인 계산 (0~3 범위 고정) let clickedLane = Math.floor(touchX / (rect.width / 4)); clickedLane = Math.max(0, Math.min(3, clickedLane)); if (clickedLane === bottomRow.lane) { // ✅ 정답 판정 bottomRow.locked = true; piano.triggerAttackRelease(bottomRow.note, 0.5, Tone.now(), 1.0); // 오타 수정: tile -> tileEl bottomRow.tileEl.classList.add('pressed'); score++; document.getElementById('score').innerText = score; // 화면 위치 기준으로 정확한 인덱스를 찾아 제거 const index = activeRows.indexOf(bottomRow); if (index > -1) { const removedRow = activeRows.splice(index, 1)[0]; // 자연스러운 사라짐 효과 후 삭제 setTimeout(() => removedRow.rowEl.remove(), 100); } // 나머지 줄들을 아래로 한 칸씩 이동 activeRows.forEach((r, idx) => { r.rowEl.style.bottom = (idx * 25) + '%'; }); currentStep++; createRow(4); // 새로운 줄 생성 // 모든 곡을 다 쳤을 때 if(currentStep >= sheet.length && activeRows.length === 0) { setTimeout(() => end(true), 300); } } else { // ❌ 오답 판정 const errTile = document.createElement('div'); errTile.className = 'tile missed'; errTile.style.left = (clickedLane * 25) + '%'; bottomRow.rowEl.appendChild(errTile); end(false); } });
